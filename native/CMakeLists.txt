cmake_minimum_required(VERSION 3.16)
project(PokerTrainerNative LANGUAGES CXX)
if(NOT EXISTS "${TEXASSOLVER_SOURCE}/src/console.cpp")
  message(FATAL_ERROR "Set TEXASSOLVER_SOURCE to the patched console checkout")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${TEXASSOLVER_SOURCE}/ext/fmt" "${CMAKE_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
set(solver_sources)
foreach(directory src src/compairer src/nodes src/ranges src/tools src/trainable src/solver src/runtime)
  file(GLOB directory_sources CONFIGURE_DEPENDS "${TEXASSOLVER_SOURCE}/${directory}/*.cpp")
  list(APPEND solver_sources ${directory_sources})
endforeach()
list(REMOVE_ITEM solver_sources "${TEXASSOLVER_SOURCE}/src/console.cpp")
add_library(TexasSolver STATIC ${solver_sources})
target_include_directories(TexasSolver PUBLIC "${TEXASSOLVER_SOURCE}/include")
target_link_libraries(TexasSolver PUBLIC fmt OpenMP::OpenMP_CXX)
add_executable(console_solver "${TEXASSOLVER_SOURCE}/src/console.cpp")
target_link_libraries(console_solver PRIVATE TexasSolver)
if(MINGW)
  target_link_options(console_solver PRIVATE -static)
endif()
